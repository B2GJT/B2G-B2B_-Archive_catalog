/* ============================================================ 빙고 게임 백엔드 API (Apps Script Web App) - 새 구글시트를 하나 만들고, 그 시트의 스크립트 편집기에 이 코드를 통째로 붙여넣으세요. - 이 시트는 다른 시트와 완전히 별개로, 빙고 전용 데이터만 저장합니다. ============================================================ */ const CONFIG = { DATA_SHEET_NAME: '빙고데이터', // 강의 목록이 들어있는 "미수강 컨텐츠" 구글시트 COURSE_LIST_SPREADSHEET_ID: '1rXG8d71mkOWKI1ChWH5WtinfRcOz1e-AhDD8HfVacfA', COURSE_LIST_SHEET_NAME: '시트1', // 실제 시트 열 위치: A=ID, B=과정명, C=카테고리(원본, 미사용), D=상태, E=수강장 링크, H=대분류, I=중분류 COURSE_LIST_COLUMNS: { TITLE: 'B', STATUS: 'D', URL: 'E', MAJOR_CATEGORY: 'H' }, COURSE_LIST_ACTIVE_VALUE: '활성', // 이 값이 아닌 행은 제외 // 강의 수가 적은 대분류는 성격이 비슷한 카테고리로 합쳐서 9칸을 항상 채울 수 있게 함 CATEGORY_MERGE_MAP: { '비즈니스 분석': '비즈니스·데이터·금융분석', '데이터': '비즈니스·데이터·금융분석', '금융·재테크': '비즈니스·데이터·금융분석', }, }; function getOrCreateSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(CONFIG.DATA_SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(CONFIG.DATA_SHEET_NAME); sheet.appendRow(['키', '이름', '전화번호뒷자리', '카테고리', '배정강의(JSON)', '클릭상태(JSON)', '생성시각', '수정시각']); } return sheet; } function letterToIndex_(letter) { let idx = 0; for (let i = 0; i < letter.length; i++) { idx = idx * 26 + (letter.toUpperCase().charCodeAt(i) - 64); } return idx - 1; } // "미수강 컨텐츠" 구글시트를 열어서 { 카테고리: [{title,url}, ...] } 형태로 반환 // - 상태가 "활성"인 행만 포함 // - 차시(에피소드) 단위 그대로 사용 (과정명 원본 그대로) // - 강의 수가 적은 대분류는 CATEGORY_MERGE_MAP에 따라 다른 카테고리로 합침 function getCourseListFromSheet_() { const ss = SpreadsheetApp.openById(CONFIG.COURSE_LIST_SPREADSHEET_ID); const sheet = ss.getSheetByName(CONFIG.COURSE_LIST_SHEET_NAME); if (!sheet) return {}; const titleIdx = letterToIndex_(CONFIG.COURSE_LIST_COLUMNS.TITLE); const statusIdx = letterToIndex_(CONFIG.COURSE_LIST_COLUMNS.STATUS); const urlIdx = letterToIndex_(CONFIG.COURSE_LIST_COLUMNS.URL); const majorIdx = letterToIndex_(CONFIG.COURSE_LIST_COLUMNS.MAJOR_CATEGORY); const data = sheet.getDataRange().getValues(); const result = {}; for (let i = 1; i < data.length; i++) { // 1행은 헤더로 간주하고 건너뜀 const status = (data[i][statusIdx] || '').toString().trim(); if (status !== CONFIG.COURSE_LIST_ACTIVE_VALUE) continue; const title = (data[i][titleIdx] || '').toString().trim(); const url = (data[i][urlIdx] || '').toString().trim(); let category = (data[i][majorIdx] || '').toString().trim(); if (!title || !category) continue; category = CONFIG.CATEGORY_MERGE_MAP[category] || category; if (!result[category]) result[category] = []; result[category].push({ title, url }); } return result; } function normalize_(str) { return (str || '').toString().trim().replace(/\s+/g, ''); } function makeKey_(name, phone, category) { return normalize_(name) + '_' + normalize_(phone) + '_' + normalize_(category); } function findRow_(sheet, key) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === key) return i + 1; // 실제 시트 행 번호(1-based) } return -1; } function jsonResponse_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { const action = e.parameter.action; // 강의 목록 조회는 참여자 정보 없이도 호출 가능 (페이지 로드 시 바로 요청) if (action === 'getCourseList') { return jsonResponse_({ categories: getCourseListFromSheet_() }); } const sheet = getOrCreateSheet_(); const name = e.parameter.name; const phone = e.parameter.phone; const category = e.parameter.category; if (!name || !phone || !category) { return jsonResponse_({ error: '이름, 전화번호, 카테고리가 모두 필요합니다.' }); } const key = makeKey_(name, phone, category); if (action === 'getBoard') { const rowNum = findRow_(sheet, key); if (rowNum > 0) { const row = sheet.getRange(rowNum, 1, 1, 8).getValues()[0]; return jsonResponse_({ categories: JSON.parse(row[4]), states: JSON.parse(row[5]), }); } return jsonResponse_({ categories: null, states: null }); } if (action === 'saveBoard') { const categories = e.parameter.categories; // JSON 문자열 그대로 저장 const states = e.parameter.states; const now = new Date(); const rowNum = findRow_(sheet, key); if (rowNum > 0) { sheet.getRange(rowNum, 5).setValue(categories); sheet.getRange(rowNum, 6).setValue(states); sheet.getRange(rowNum, 8).setValue(now); } else { sheet.appendRow([key, name, phone, category, categories, states, now, now]); } return jsonResponse_({ ok: true }); } if (action === 'setCell') { const index = parseInt(e.parameter.index, 10); const value = e.parameter.value === '1'; const rowNum = findRow_(sheet, key); if (rowNum < 0) return jsonResponse_({ error: '해당 참여자의 빙고판을 찾을 수 없습니다. getBoard/saveBoard를 먼저 호출하세요.' }); const statesCell = sheet.getRange(rowNum, 6); const states = JSON.parse(statesCell.getValue()); if (index < 0 || index >= states.length) return jsonResponse_({ error: '잘못된 칸 번호입니다.' }); states[index] = value; statesCell.setValue(JSON.stringify(states)); sheet.getRange(rowNum, 8).setValue(new Date()); return jsonResponse_({ ok: true, states: states }); } return jsonResponse_({ error: '알 수 없는 action 입니다.' }); } catch (err) { return jsonResponse_({ error: err.toString() }); } }
